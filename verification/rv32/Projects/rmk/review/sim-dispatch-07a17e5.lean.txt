def memT (t : List (Nat × Nat)) (a : Word) : BitVec 8 :=
  match t.find? (fun p => p.1 == a.toNat) with
  | some p => BitVec.ofNat 8 p.2
  | none => 0
def mkCore (pc : Word) (rs : List (Nat × Word)) : Core where
  pc := pc
  x := fun r => match rs.find? (fun p => p.1 == r.val) with | some p => p.2 | none => 0
  mie_ := false
  mpie := false
  mpp := 3
  mieReg := 0
  mtvec := 0x800000f8
  mepc := 0
  mcause := 0
  mscratch := 0
  resv := false
partial def sim (m : Machine) (n : Nat) (acc : List String) : List String :=
  if n = 0 then ("FUEL" :: acc).reverse else
  match fetchDecode img m.core.pc with
  | none => (s!"NOFETCH {m.core.pc.toNat}" :: acc).reverse
  | some (i, len) =>
    let e := exec m.core len i
    let note := match e with
      | .load w a _ _ => if B.inRam a w.bytes then "" else s!" ROM {a.toNat} {w.bytes}"
      | _ => ""
    let acc := s!"{m.core.pc.toNat}{note}" :: acc
    match stepInstr B img m (fun _ => 0) with
    | .fault f => (s!"FAULT {repr f}" :: acc).reverse
    | .ok c r _ => sim ⟨c, r⟩ (n - 1) acc
